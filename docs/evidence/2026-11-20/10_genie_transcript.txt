WITH rule_cte AS (
  SELECT
    MAX(`intelligence_alert_rules`.`threshold`) AS validated_limit_c,
    MAX(`intelligence_alert_rules`.`fault_code`) AS fault_code,
    MAX(`intelligence_alert_rules`.`severity`) AS rule_severity
  FROM <catalog>.`pharma_twin`.`intelligence_alert_rules`
  WHERE `intelligence_alert_rules`.`industry` = 'manufacturing'
    AND `intelligence_alert_rules`.`rule_id` = 'mfg-alert-spindle-temp'
), sensor_cte AS (
  SELECT MAX(`intelligence_sensor_bindings`.`sensor_id`) AS sensor_id
  FROM <catalog>.`pharma_twin`.`intelligence_sensor_bindings`
  WHERE `intelligence_sensor_bindings`.`industry` = 'manufacturing'
    AND `intelligence_sensor_bindings`.`sensor_name` = 'cnc_1_spindle_temp'
), over_limit AS (
  SELECT
    `lb__shadow_manufacturing_sensor_readings_history`.`event_time`,
    `lb__shadow_manufacturing_sensor_readings_history`.`metadata`:lot_id::string AS lot_id,
    `lb__shadow_manufacturing_sensor_readings_history`.`metadata`:part_serial::string AS part_serial,
    `lb__shadow_manufacturing_sensor_readings_history`.`value`,
    rule_cte.validated_limit_c,
    CASE
      WHEN LAG(`lb__shadow_manufacturing_sensor_readings_history`.`event_time`) OVER (ORDER BY `lb__shadow_manufacturing_sensor_readings_history`.`event_time`) IS NULL THEN 1
      WHEN `lb__shadow_manufacturing_sensor_readings_history`.`event_time` > LAG(`lb__shadow_manufacturing_sensor_readings_history`.`event_time`) OVER (ORDER BY `lb__shadow_manufacturing_sensor_readings_history`.`event_time`) + INTERVAL 5 MINUTES THEN 1
      ELSE 0
    END AS new_excursion_flag
  FROM <catalog>.`pharma_twin`.`lb__shadow_manufacturing_sensor_readings_history`
  CROSS JOIN rule_cte
  WHERE `lb__shadow_manufacturing_sensor_readings_history`.`_pg_change_type` IN ('insert', 'update_postimage')
    AND `lb__shadow_manufacturing_sensor_readings_history`.`industry` = 'manufacturing'
    AND `lb__shadow_manufacturing_sensor_readings_history`.`sensor_name` = 'cnc_1_spindle_temp'
    AND `lb__shadow_manufacturing_sensor_readings_history`.`event_time` IS NOT NULL
    AND `lb__shadow_manufacturing_sensor_readings_history`.`event_time` >= current_timestamp() - INTERVAL 24 HOURS
    AND `lb__shadow_manufacturing_sensor_readings_history`.`value` > rule_cte.validated_limit_c
), excursion_rows AS (
  SELECT
    over_limit.event_time,
    over_limit.lot_id,
    over_limit.part_serial,
    over_limit.value,
    over_limit.validated_limit_c,
    SUM(over_limit.new_excursion_flag) OVER (ORDER BY over_limit.event_time ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS excursion_id
  FROM over_limit
), excursion_summary AS (
  SELECT
    excursion_rows.excursion_id,
    MIN(excursion_rows.event_time) AS first_over_limit_at,
    MAX(excursion_rows.event_time) AS last_over_limit_at,
    MAX(excursion_rows.value) AS peak_temp_c,
    MAX(excursion_rows.validated_limit_c) AS validated_limit_c,
    COUNT(DISTINCT excursion_rows.part_serial) AS parts_at_risk,
    CONCAT_WS(', ', SORT_ARRAY(COLLECT_SET(excursion_rows.lot_id))) AS lots_at_risk

  WHERE excursion_rows.lot_id IS NOT NULL
    AND excursion_rows.part_serial IS NOT NULL
  GROUP BY excursion_rows.excursion_id
), alert_candidates AS (
  SELECT
    excursion_summary.excursion_id,
    `intelligence_alert_snapshot`.`alert_id`,
    `intelligence_alert_snapshot`.`status`,
    `intelligence_alert_snapshot`.`severity`,
    `intelligence_alert_snapshot`.`fault_code`,
    TO_TIMESTAMP(`intelligence_alert_snapshot`.`opened_at`) AS opened_at,
    TO_TIMESTAMP(`intelligence_alert_snapshot`.`updated_at`) AS updated_at,
    ROW_NUMBER() OVER (

  FROM excursion_summary
  CROSS JOIN sensor_cte
  CROSS JOIN rule_cte
  LEFT JOIN <catalog>.`pharma_twin`.`intelligence_alert_snapshot`
    ON `intelligence_alert_snapshot`.`industry` = 'manufacturing'
   AND `intelligence_alert_snapshot`.`sensor_id` = sensor_cte.sensor_id
   AND `intelligence_alert_snapshot`.`rule_id` = 'mfg-alert-spindle-temp'
   AND `intelligence_alert_snapshot`.`fault_code` = rule_cte.fault_code
   AND TO_TIMESTAMP(`intelligence_alert_snapshot`.`opened_at`) >= excursion_summary.first_over_limit_at - INTERVAL 30 MINUTES
   AND TO_TIMESTAMP(`intelligence_alert_snapshot`.`opened_at`) <= excursion_summary.last_over_limit_at + INTERVAL 30 MINUTES
)

  alert_candidates.fault_code,
  alert_candidates.opened_at AS alert_opened_at,
  alert_candidates.updated_at AS alert_updated_at,

    ELSE 'review_no_parts_identified'
  END AS ncr_recommendation
FROM excursion_summary
CROSS JOIN rule_cte
LEFT JOIN alert_candidates
  ON excursion_summary.excursion_id = alert_candidates.excursion_id
 AND alert_candidates.rn = 1
ORDER BY excursion_summary.first_over_limit_at DESC
